// REGLA: el seed de 10 vehículos NO se altera; el bloque 5-7 usa un
// documento temporal "PruebaDemo" que se crea y se borra solo.

// ---- 1) Ver todos (equivale a GET /vehiculos) ----
db.vehiculos.find({}, { _id: 0 }).sort({ nombre: 1 }).toArray();

// ---- 2) Consultar uno por nombre (equivale a GET /vehiculos/<nombre>) ----
// C1 (caché 15s, demo Integrante 1):
db.vehiculos.findOne({ nombre: "Avión" }, { _id: 0 });
// C4 (NO_CACHE, demo Integrante 1):
db.vehiculos.findOne({ nombre: "Automóvil" }, { _id: 0 });

// ---- 3) Conteo por categoría (evidencia de estructura, punto 2 del video) ----
db.vehiculos.aggregate([
  { $group: { _id: "$categoria", total: { $sum: 1 }, peso_promedio_t: { $avg: "$peso_toneladas" } } },
  { $sort: { _id: 1 } }
]).toArray();

// ---- 4) Matriz de caché por CATEGORÍA (lógica espejo de la API) ----
// Campo identificador: categoria. Bandas: <=5 -> 15s | 6-10 -> 20s |
// 11-20 -> 40s | >20 -> 0 = NO_CACHE (prioritaria).
db.vehiculos.aggregate([
  { $project: {
      _id: 0, nombre: 1, categoria: 1,
      len_categoria: { $strLenCP: "$categoria" },
      ttl_esperado_s: { $switch: {
        branches: [
          { case: { $gt: [{ $strLenCP: "$categoria" }, 20] }, then: 0 },
          { case: { $lte: [{ $strLenCP: "$categoria" }, 5] }, then: 15 },
          { case: { $lte: [{ $strLenCP: "$categoria" }, 10] }, then: 20 }
        ],
        default: 40
      } }
  } },
  { $sort: { ttl_esperado_s: 1, nombre: 1 } }
]).toArray();
// Esperado con seed intacto: 15s Avión/Helicóptero | 20s Barco/Lancha/Cohete |
// 0 (NO_CACHE) Automóvil/Motocicleta/Camión/Bicicleta/Monopatín.
// La banda 40s (11-20) no existe en el seed -> ver DB/03_dato_prueba_c3.txt.

// ---- 5) POST equivalente: registrar (documento TEMPORAL, se borra en 7) ----
db.vehiculos.insertOne({ nombre: "PruebaDemo", categoria: "Aéreo", peso_toneladas: 1, descripcion: "Avión de prueba." });

// ---- 6) PUT equivalente: actualizar ----
db.vehiculos.updateOne({ nombre: "PruebaDemo" }, { $set: { peso_toneladas: 2 } });

// ---- 7) DELETE equivalente: eliminar + verificación de seed intacto ----
db.vehiculos.deleteOne({ nombre: "PruebaDemo" });
db.vehiculos.countDocuments(); // debe volver a 10
